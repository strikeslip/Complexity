ENHANCEMENT PROPOSALS ### A. TRUE GRANULAR SYNTHESIS ENGINE The current oscillator approach misses the sonic character of actual seismic waveforms. Here's a complete implementation for true sample-based granular synthesis: ```javascript // ═══════════════════════════════════════════════════════════════════════════ // ENHANCED GRANULAR SYNTHESIS ENGINE // ═══════════════════════════════════════════════════════════════════════════ class SeismicGranulator { constructor(audioCtx, seismicBuffer, config = {}) { this.ctx = audioCtx; this.buffer = seismicBuffer; // Grain parameters with sensible defaults this.config = { grainSize: config.grainSize || 0.08, // 80ms default grainSizeVar: config.grainSizeVar || 0.3, // ±30% variation density: config.density || 20, // grains/second pitchShift: config.pitchShift || 1.0, // playback rate pitchVar: config.pitchVar || 0.1, // ±10% variation positionSpread: config.positionSpread || 0.2, panSpread: config.panSpread || 0.8, reverse: config.reverse || false, windowType: config.windowType || 'hanning', // 'hanning', 'gaussian', 'tukey' attackRatio: config.attackRatio || 0.35, // for tukey window ...config }; // Pre-computed window functions this.windowCache = new Map(); // Grain cloud state this.activeGrains = new Set(); this.grainPool = []; this.poolSize = 100; this.initGrainPool(); } initGrainPool() { // Pre-allocate grain nodes for efficiency for (let i = 0; i < this.poolSize; i++) { this.grainPool.push({ source: null, gain: this.ctx.createGain(), panner: this.ctx.createStereoPanner(), inUse: false }); } } getWindow(size, type = 'hanning') { const key = `${size}-${type}`; if (this.windowCache.has(key)) return this.windowCache.get(key); const window = new Float32Array(size); switch(type) { case 'hanning': for (let i = 0; i < size; i++) { window[i] = 0.5 * (1 - Math.cos(2 * Math.PI * i / (size - 1))); } break; case 'gaussian': const sigma = 0.4; for (let i = 0; i < size; i++) { const n = (i - (size - 1) / 2) / (sigma * (size - 1) / 2); window[i] = Math.exp(-0.5 * n * n); } break; case 'tukey': const alpha = this.config.attackRatio * 2; for (let i = 0; i < size; i++) { const x = i / (size - 1); if (x < alpha / 2) { window[i] = 0.5 * (1 + Math.cos(Math.PI * (2 * x / alpha - 1))); } else if (x > 1 - alpha / 2) { window[i] = 0.5 * (1 + Math.cos(Math.PI * (2 * x / alpha - 2 / alpha + 1))); } else { window[i] = 1; } } break; case 'expodec': // Exponential decay - sharp attack, gradual release for (let i = 0; i < size; i++) { const attack = Math.min(1, i / (size * 0.1)); const decay = Math.exp(-3 * i / size); window[i] = attack * decay; } break; } this.windowCache.set(key, window); return window; } scheduleGrain(time, params = {}) { const cfg = { ...this.config, ...params }; // Find available grain from pool let grain = this.grainPool.find(g => !g.inUse); if (!grain) return; // Pool exhausted grain.inUse = true; this.activeGrains.add(grain); // Calculate grain parameters const sizeVar = 1 + (Math.random() - 0.5) * 2 * cfg.grainSizeVar; const grainDuration = cfg.grainSize * sizeVar; const grainSamples = Math.floor(grainDuration * this.ctx.sampleRate); // Position in buffer with spread const basePosition = cfg.position !== undefined ? cfg.position : (this.currentPosition || 0); const posOffset = (Math.random() - 0.5) * cfg.positionSpread; let startSample = Math.floor((basePosition + posOffset) * this.buffer.length); startSample = Math.max(0, Math.min(this.buffer.length - grainSamples, startSample)); // Create grain buffer with window applied const grainBuffer = this.ctx.createBuffer( this.buffer.numberOfChannels, grainSamples, this.ctx.sampleRate ); const window = this.getWindow(grainSamples, cfg.windowType); for (let ch = 0; ch < this.buffer.numberOfChannels; ch++) { const sourceData = this.buffer.getChannelData(ch); const grainData = grainBuffer.getChannelData(ch); for (let i = 0; i < grainSamples; i++) { const readIndex = cfg.reverse ? startSample + grainSamples - 1 - i : startSample + i; const sample = sourceData[readIndex] || 0; grainData[i] = sample * window[i]; } } // Create and configure source grain.source = this.ctx.createBufferSource(); grain.source.buffer = grainBuffer; // Pitch with variation const pitchVar = 1 + (Math.random() - 0.5) * 2 * cfg.pitchVar; grain.source.playbackRate.setValueAtTime(cfg.pitchShift * pitchVar, time); // Amplitude const amplitude = cfg.amplitude || 0.3; grain.gain.gain.setValueAtTime(amplitude * (0.7 + Math.random() * 0.3), time); // Pan const pan = (Math.random() - 0.5) * 2 * cfg.panSpread; grain.panner.pan.setValueAtTime(pan, time); // Connect grain.source.connect(grain.gain); grain.gain.connect(grain.panner); grain.panner.connect(cfg.destination || this.ctx.destination); // Optional effects sends if (cfg.reverbSend) { const reverbGain = this.ctx.createGain(); reverbGain.gain.value = cfg.reverbAmount || 0.3; grain.gain.connect(reverbGain); reverbGain.connect(cfg.reverbSend); } // Schedule const actualDuration = grainDuration / (cfg.pitchShift * pitchVar); grain.source.start(time); grain.source.stop(time + actualDuration + 0.01); // Cleanup grain.source.onended = () => { grain.inUse = false; this.activeGrains.delete(grain); try { grain.source.disconnect(); } catch(e) {} }; } // Schedule multiple grains using look-ahead scheduleGrainCloud(startTime, duration, params = {}) { const cfg = { ...this.config, ...params }; const interval = 1 / cfg.density; let time = startTime; while (time < startTime + duration) { const jitter = interval * 0.3 * (Math.random() - 0.5); this.scheduleGrain(time, cfg); time += interval + jitter; } } setPosition(position) { this.currentPosition = Math.max(0, Math.min(1, position)); } updateConfig(newConfig) { Object.assign(this.config, newConfig); } } ``` #### Integration with Attractor System ```javascript // In updateSynthesisParams(): if (granulator) { granulator.updateConfig({ grainSize: blended.grainDuration, density: blended.density, pitchShift: blended.frequencyMult, pitchVar: blended.spread * 0.5, positionSpread: blended.spread * 0.3, windowType: getWindowForAttractor(currentAttractor) }); granulator.setPosition(seismicPlayhead); } function getWindowForAttractor(attractor) { const windowMap = { 'Drone': 'gaussian', // Smooth, overlapping 'Texture': 'hanning', // Balanced 'Pulse': 'tukey', // Punchy attack 'Sparse': 'expodec' // Percussive decay }; return windowMap[attractor?.name] || 'hanning'; } ``` ### B. this.ctx = canvas.getContext('2d'); this.config = { // Earth model earthRadius: Math.min(canvas.width, canvas.height) * 0.35, coreRatio: 0.545, // Outer core radius / Earth radius innerCoreRatio: 0.19, // Inner core radius / Earth radius // Shadow zone angles (degrees from epicenter) shadowStart: 103, shadowEnd: 142, // Colors mantleColor: '#1a2a3a', outerCoreColor: '#4a2a1a', innerCoreColor: '#6a4a2a', crustColor: '#2d4a5a', shadowColor: 'rgba(0, 0, 0, 0.6)', pWaveColor: '#4a9eff', sWaveColor: '#ff6a4a', surfaceWaveColor: '#4aff9e', // Animation waveSpeed: 0.02, rayCount: 24, ...config }; this.epicenterAngle = -Math.PI / 2; // Top of Earth this.animationPhase = 0; this.seismicAmplitude = 0; this.rays = []; this.initRays(); } initRays() { // Pre-calculate ray paths through Earth this.rays = []; const deltaAngle = Math.PI / (this.config.rayCount - 1); for (let i = 0; i < this.config.rayCount; i++) { const takeoffAngle = -Math.PI / 2 + i * deltaAngle; const ray = this.calculateRayPath(takeoffAngle); this.rays.push(ray); } } calculateRayPath(takeoffAngle) { // Simplified ray tracing through Earth's layers // In reality, Snell's law would be applied at each boundary const R = this.config.earthRadius; const coreR = R * this.config.coreRatio; const innerR = R * this.config.innerCoreRatio; const points = []; const cx = this.canvas.width / 2; const cy = this.canvas.height / 2; // Start at epicenter const startX = cx + R * Math.cos(this.epicenterAngle); const startY = cy + R * Math.sin(this.epicenterAngle); points.push({ x: startX, y: startY, layer: 'surface' }); // Ray direction const dirX = Math.cos(takeoffAngle); const dirY = Math.sin(takeoffAngle); // Check intersections with core const dx = startX - cx; const dy = startY - cy; // Parametric ray-circle intersection const a = dirX * dirX + dirY * dirY; const b = 2 * (dx * dirX + dy * dirY); // Check outer core intersection const cCore = dx * dx + dy * dy - coreR * coreR; const discCore = b * b - 4 * a * cCore; let hitsCore = false; let coreEntry = null; let coreExit = null; if (discCore > 0) { const t1 = (-b - Math.sqrt(discCore)) / (2 * a); const t2 = (-b + Math.sqrt(discCore)) / (2 * a); if (t1 > 0 && t2 > 0) { hitsCore = true; coreEntry = { x: startX + dirX * t1, y: startY + dirY * t1, layer: 'core-entry' }; coreExit = { x: startX + dirX * t2, y: startY + dirY * t2, layer: 'core-exit' }; } } if (hitsCore) { // Ray enters core - gets refracted (simplified as straight path through core) points.push(coreEntry); points.push(coreExit); // Calculate exit point on far side of Earth // Refraction causes the shadow zone const exitAngle = Math.atan2(coreExit.y - cy, coreExit.x - cx); const surfaceExit = { x: cx + R * Math.cos(exitAngle + Math.PI * 0.2), // Refraction bends ray y: cy + R * Math.sin(exitAngle + Math.PI * 0.2), layer: 'surface', refracted: true }; points.push(surfaceExit); } else { // Ray travels through mantle only // Find exit point on opposite side const cSurface = dx * dx + dy * dy - R * R; const discSurface = b * b - 4 * a * cSurface; if (discSurface > 0) { const t = (-b + Math.sqrt(discSurface)) / (2 * a); points.push({ x: startX + dirX * t, y: startY + dirY * t, layer: 'surface' }); } } return { points, hitsCore, takeoffAngle }; } setEpicenter(lon, lat) { // Convert geographic coordinates to canvas angle this.epicenterAngle = -Math.PI / 2 + (lon / 180) * Math.PI; this.initRays(); } setSeismicAmplitude(amplitude) { this.seismicAmplitude = Math.max(0, Math.min(1, amplitude)); } draw(phase = 0) { const ctx = this.ctx; const w = this.canvas.width; const h = this.canvas.height; const cx = w / 2; const cy = h / 2; const R = this.config.earthRadius; // Clear ctx.fillStyle = '#08080c'; ctx.fillRect(0, 0, w, h); // Draw Earth layers this.drawEarthLayers(cx, cy, R); // Draw shadow zones this.drawShadowZones(cx, cy, R); // Draw seismic rays this.drawSeismicRays(cx, cy, R, phase); // Draw epicenter this.drawEpicenter(cx, cy, R); // Draw station indicator this.drawStation(cx, cy, R); // Update phase this.animationPhase = phase; } drawEarthLayers(cx, cy, R) { const ctx = this.ctx; // Mantle const mantleGradient = ctx.createRadialGradient(cx, cy, 0, cx, cy, R); mantleGradient.addColorStop(0, '#0a1520'); mantleGradient.addColorStop(0.55, this.config.mantleColor); mantleGradient.addColorStop(1, this.config.crustColor); ctx.fillStyle = mantleGradient; ctx.beginPath(); ctx.arc(cx, cy, R, 0, Math.PI * 2); ctx.fill(); // Outer core const coreR = R * this.config.coreRatio; const coreGradient = ctx.createRadialGradient(cx, cy, 0, cx, cy, coreR); coreGradient.addColorStop(0, '#8a5a3a'); coreGradient.addColorStop(1, this.config.outerCoreColor); ctx.fillStyle = coreGradient; ctx.beginPath(); ctx.arc(cx, cy, coreR, 0, Math.PI * 2); ctx.fill(); // Inner core const innerR = R * this.config.innerCoreRatio; ctx.fillStyle = this.config.innerCoreColor; ctx.beginPath(); ctx.arc(cx, cy, innerR, 0, Math.PI * 2); ctx.fill(); // Core boundary glow ctx.strokeStyle = 'rgba(255, 150, 100, 0.3)'; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(cx, cy, coreR, 0, Math.PI * 2); ctx.stroke(); } drawShadowZones(cx, cy, R) { const ctx = this.ctx; // Convert shadow zone angles to radians const shadowStart = this.config.shadowStart * Math.PI / 180; const shadowEnd = this.config.shadowEnd * Math.PI / 180; // Draw both shadow zones (symmetric) for (let side = 0; side < 2; side++) { const startAngle = this.epicenterAngle + (side === 0 ? shadowStart : -shadowEnd); const endAngle = this.epicenterAngle + (side === 0 ? shadowEnd : -shadowStart); // Shadow zone arc ctx.fillStyle = this.config.shadowColor; ctx.beginPath(); ctx.moveTo(cx, cy); ctx.arc(cx, cy, R * 1.3, startAngle, endAngle); ctx.closePath(); ctx.fill(); // Shadow zone label const midAngle = (startAngle + endAngle) / 2; const labelR = R * 1.15; ctx.fillStyle = 'rgba(255, 100, 100, 0.7)'; ctx.font = '10px monospace'; ctx.textAlign = 'center'; ctx.fillText( 'SHADOW ZONE', cx + labelR * Math.cos(midAngle), cy + labelR * Math.sin(midAngle) ); } } drawSeismicRays(cx, cy, R, phase) { const ctx = this.ctx; return { x: ray.points[i-1].x + dx * t, y: ray.points[i-1].y + dy * t }; } traveled += segLength; } return ray.points[ray.points.length - 1]; } drawEpicenter(cx, cy, R) { const ctx = this.ctx; const epicX = cx + R * Math.cos(this.epicenterAngle); ctx.arc(epicX, epicY, pulseSize * 2, 0, Math.PI * 2); ctx.fill(); // Epicenter marker ctx.fillStyle = '#ff4444'; ctx.beginPath(); ctx.arc(epicX, epicY, 5, 0, Math.PI * 2); ctx.fill(); // Label ctx.fillStyle = '#ff8888'; ctx.font = '9px monospace'; ctx.textAlign = 'center'; ctx.fillText('EPICENTER', epicX, epicY - 15); } drawStation(cx, cy, R) { const ctx = this.ctx; // Station at a fixed angular distance (e.g., 80° - within good reception zone) const stationAngle = this.epicenterAngle + Math.PI * 0.44; // ~80° const stationX = cx + R * Math.cos(stationAngle); const stationY = cy + R * Math.sin(stationAngle); // Station marker ctx.fillStyle = '#4aff9e'; ctx.beginPath(); ctx.arc(stationX, stationY, 4, 0, Math.PI * 2);
